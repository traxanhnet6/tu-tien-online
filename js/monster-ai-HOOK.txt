2) engine.js — hàm _updateMonsters(dt, ps) (khoảng dòng 2218),
   vòng for từng quái, ngay sau dòng 2223:

   và TRƯỚC khối quán tính giật lùi dòng 2224:
     if (m.kx || m.ky) { ...
   Chèn đúng đoạn này:
     /* MonsterAI 2026-10-27: AI quái mới (lang thang/truy đuổi/
      * quay về bãi) — xem js/monster-ai.js. Gọi TRƯỚC khối kx/ky
      * vì updateOne tự xóa kx/ky nên quái không bị hất văng. */
     var _aiDone = false;
     try {
       if (typeof MonsterAI !== 'undefined' && MonsterAI.updateOne)
         _aiDone = !!MonsterAI.updateOne(m, player, dt, { mapId: _mapId,
           zones: (_mapDef && _mapDef.spawnZones) || null,
           monsters: _monsters, worldW: WORLD_W, worldH: WORLD_H });
     } catch (_e) {}
   (updateOne trả true = đã xử lý di chuyển; false = quái chết/
   choáng/sợ hãi — để engine xử lý như cũ. Nếu MonsterAI chưa
   nạp, typeof guard giữ game chạy AI cũ, không crash.)
